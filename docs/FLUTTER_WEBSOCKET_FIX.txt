El frontend está construyendo manualmente la URL del WebSocket en lugar de usar 
la URL que el backend devuelve en /session/init response.

```dart
Future<void> connectWebSocket(String websocketUrl) async {
  // Usar la URL que viene del backend
  _channel = WebSocketChannel.connect(Uri.parse(websocketUrl));
  
  // Escuchar mensajes
  _channel!.stream.listen((message) {
    final state = jsonDecode(message as String);
    _stateController.add(state);
  }, onError: (error) {
    print('❌ WebSocket error: $error');
  }, onDone: () {
    print('🔌 WebSocket closed');
  });
  
  // ⚡ CRÍTICO: Enviar comando PLAY inmediatamente después de conectar
  _sendControlMessage('play');
}

// Agregar este método
void _sendControlMessage(String action) {
  if (_channel != null) {
    _channel!.sink.add(jsonEncode({
      "type": "control",
      "action": action,
    }));
    print('▶️  Sent $action control message');
  }
}
```

```dart
  final sessionData = await _initSession();
  if (sessionData != null) {
    _sessionId = sessionData['session_id'];
    _websocketUrl = sessionData['websocket_url'];  // ✅ Obtener URL del backend

Future<Map<String, dynamic>?> _initSession() async {
  try {
    final response = await http.post(
      Uri.parse(BridgeConfig.sessionInitUrl),
      headers: {"Content-Type": "application/json"},
      body: jsonEncode({
        "preferences": {
          "tts_enabled": true,
          "auto_translate": true,  
          "target_language": "en",
          "word_pause_ms": 4000,
          "phrase_pause_ms": 8000,
        },
      }),
    );

    print('📡 Session init response: ${response.statusCode}');

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      print('✅ Session initialized:');
      print('   session_id: ${data['session_id']}');
      print('   websocket_url: ${data['websocket_url']}');  // ✅ Verificar que existe
      
      return {
        'session_id': data['session_id'],
        'websocket_url': data['websocket_url'],  // ✅ Retornar ambos
      };
    }
  } catch (e) {
    print('❌ Error initializing session: $e');
  }
  return null;
}
```

```dart
Future<String> _convertCameraImageToBase64(CameraImage image) async {
  try {
    // Convertir YUV a RGB
    final bytes = image.planes[0].bytes;
    
    // Crear imagen en formato JPEG
    final img.Image? convertedImage = img.decodeImage(bytes);
    if (convertedImage == null) {
      throw Exception('Failed to decode image');
    }
    
    // Codificar a JPEG con calidad 90
    final jpegBytes = img.encodeJpg(convertedImage, quality: 90);
    
    // Convertir a base64
    final base64String = base64Encode(jpegBytes);
    
    print('📸 Frame encoded: ${base64String.length} chars');
    
    return base64String;
  } catch (e) {
    print('❌ Error encoding frame: $e');
    rethrow;
  }
}
```

4. AGREGAR: Listener para estado en HomeScreen
----------------------------------------------------------------

```dart
void _setupStateListener() {
  _sessionService.stateStream.listen((state) {
    print('📨 State update received:');
    print('   detection.letter: ${state['detection']?['letter']}');
    print('   word.corrected: ${state['word']?['corrected']}');
    print('   sentence.current: ${state['sentence']?['current']}');
    print('   sentence.just_completed: ${state['sentence']?['just_completed']}');
    
    setState(() {
      detectedLetter = state['detection']?['letter'] ?? '';
      correctedWord = state['word']?['corrected'] ?? '';
      sentence = state['sentence']?['current'] ?? '';
      translation = state['translation']?['translated_sentence'] ?? '';
    });
    
    // ⚡ CRÍTICO: Reproducir TTS cuando se completa la oración
    final justCompleted = state['sentence']?['just_completed'] as bool? ?? false;
    if (justCompleted) {
      final completedText = state['sentence']?['completed'] as String? ?? '';
      if (completedText.isNotEmpty) {
        print('🔊 Sentence completed: $completedText');
        _playTTS(state);
      }
    }
  });
}

Future<void> _playTTS(Map<String, dynamic> state) async {
  final tts = state['tts'] as Map<String, dynamic>?;
  if (tts == null) return;
  
  final audioBase64 = tts['audio_base64'] as String?;
  if (audioBase64 == null || audioBase64.isEmpty) {
    print('⚠️  No TTS audio available');
    return;
  }
  
  try {
    // Decodificar base64
    final audioBytes = base64Decode(audioBase64);
    
    // Guardar a archivo temporal
    final tempDir = await getTemporaryDirectory();
    final tempFile = File('${tempDir.path}/tts_${DateTime.now().millisecondsSinceEpoch}.mp3');
    await tempFile.writeAsBytes(audioBytes);
    
    // Reproducir con audioplayers
    final player = AudioPlayer();
    await player.play(DeviceFileSource(tempFile.path));
    
    print('🔊 TTS audio playing');
    
    // Limpiar después de reproducir
    player.onPlayerComplete.listen((_) {
      tempFile.delete();
      player.dispose();
    });
    
  } catch (e) {
    print('❌ Error playing TTS: $e');
  }
}

@override
void initState() {
  super.initState();
  _sessionService = BridgeSessionService();
  _setupStateListener();  // ✅ Configurar listener
}
```
